
use appointment_db;

//User Collection
db.createCollection("users", {
   validator: {
      $jsonSchema: {
         bsonType: "object",
         required: ["username", "password"],
         properties: {
            username: { bsonType: "string" },
            password: { bsonType: "string" },
            full_name: { bsonType: "string" },
            nickname: { bsonType: "string" },
            phone: { bsonType: "string" },
            image_url: { bsonType: "string" },
            created_at: { bsonType: "date" },
            role: { bsonType: "string" }
         }
      }
   }
});
db.users.createIndex({ username: 1 }, { unique: true });

// Pets Collection
db.createCollection("pets", {
   validator: {
      $jsonSchema: {
         bsonType: "object",
         required: ["pet_name", "species"],
         properties: {
            user_id: { bsonType: "objectId" }, // References users._id
            pet_name: { bsonType: "string" },
            species: { bsonType: "string" },
            bloodtype: { bsonType: "string" },
            birth_date: { bsonType: "date" },
            weight: { bsonType: "decimal" },
            allergy: { bsonType: "string" },
            breed: { bsonType: "string" },
            gender: { bsonType: "string" },
            image_url: { bsonType: "string" }
         }
      }
   }
});

// Doctor Collection
db.createCollection("doctors", {
   validator: {
      $jsonSchema: {
         bsonType: "object",
         properties: {
            user_id: { bsonType: "objectId" }, // References users._id
            specialization: { bsonType: "string" },
            work_time: { bsonType: "string" },
            is_available: { bsonType: "string" }
         }
      }
   }
});

// Appointments Collection
db.createCollection("appointments", {
   validator: {
      $jsonSchema: {
         bsonType: "object",
         properties: {
            user_id: { bsonType: "objectId" }, // References users._id
            pet_id: { bsonType: "objectId" },  // References pets._id
            doc_id: { bsonType: "objectId" },  // References doctors._id
            reason: { bsonType: "string" },
            status: { bsonType: "string" }, // Default logic handled in application layer, or custom validator
            app_date: { bsonType: "date" }  // Combined date and time into one Date object is recommended
         }
      }
   }
});

// MedicalRecords Collection
db.createCollection("medical_records", {
   validator: {
      $jsonSchema: {
         bsonType: "object",
         properties: {
            pet_id: { bsonType: "objectId" },
            doc_id: { bsonType: "objectId" },
            app_id: { bsonType: "objectId" },
            diagnosis: { bsonType: "string" },
            treatment_detail: { bsonType: "string" },
            cost: { bsonType: "decimal" },
            treatment_date: { bsonType: "date" }
         }
      }
   }
});

// Fact_table Collection
db.createCollection("fact_table", {
   validator: {
      $jsonSchema: {
         bsonType: "object",
         properties: {
            app_id: { bsonType: "objectId" },
            user_id: { bsonType: "objectId" },
            pet_id: { bsonType: "objectId" },
            doc_id: { bsonType: "objectId" },
            record_id: { bsonType: "objectId" }
         }
      }
   }
});
